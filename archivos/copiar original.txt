@echo off
title Instalar Librería ONPE
color 0A

set "ORIGEN=%~dp0iofe.indecopi.gob.pe_TSL_tsl-pe.xml"
set "DESTINO=C:\Users\FN\.firmaONPE\cache"

:: Crear la carpeta si no existe
if not exist "%DESTINO%" (
    mkdir "%DESTINO%"
)

:: Eliminar todo el contenido de la carpeta cache
del /Q "%DESTINO%\*.*" 2>nul

:: Eliminar subcarpetas (si las hubiera)
for /d %%i in ("%DESTINO%\*") do rd /S /Q "%%i"

:: Copiar el nuevo archivo
copy "%ORIGEN%" "%DESTINO%" /Y

echo.
echo ==========================================
echo La carpeta cache fue limpiada.
echo El archivo se copio correctamente.
echo ==========================================
pause
